# One group on purpose. robots.txt groups do NOT merge: a crawler obeys the
# most specific group that names it and ignores every other one (RFC 9309). So
# adding a friendly "User-agent: GPTBot / Allow: /" block does not add a
# welcome — it REMOVES the two disallows below for that crawler, and quietly
# opens /admin and /api to it. This file welcomes every assistant already, by
# welcoming everyone.
User-agent: *
Allow: /
# The operator dashboard and the sync API are not content.
Disallow: /admin
Disallow: /api/
